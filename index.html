import requests import kivy from kivy.app import App from kivy.uix.boxlayout import BoxLayout from kivy.uix.gridlayout import GridLayout from kivy.uix.scrollview import ScrollView from kivy.uix.label import Label from kivy.uix.button import Button from kivy.uix.textinput import TextInput from kivy.uix.image import AsyncImage from kivy.uix.screenmanager import ScreenManager, Screen from kivy.uix.behaviors import ButtonBehavior from kivy.uix.popup import Popup from kivy.uix.progressbar import ProgressBar from kivy.metrics import dp from kivy.utils import get_color_from_hex from kivy.storage.jsonstore import JsonStore from kivy.graphics import Color, RoundedRectangle, Rectangle import webbrowser kivy.require("2.1.0") # --------------------------- def parse_books(data): """Extrahiert Bücher aus der Google Books API Antwort und fügt Affiliate-Links hinzu.""" books = [] for item in data.get("items", []): vol = item["volumeInfo"] sale = item.get("saleInfo", {}) list_price = sale.get("listPrice", {}) # Preis ermitteln if list_price.get("amount"): price = f"{list_price['amount']} {list_price.get('currencyCode', '€')}" else: price = "Preis nicht verfügbar" # ISBN ermitteln (für Affiliate-Link) isbn = "" for id_obj in vol.get("industryIdentifiers", []): if id_obj["type"] in ["ISBN_13", "ISBN_10"]: isbn = id_obj["identifier"] break title = vol.get("title", "Unbekannter Titel") books.append({ "id": item["id"], "title": title, "author": ", ".join(vol.get("authors", ["Unbekannt"])), "price": price, "cover": vol.get("imageLinks", {}).get("thumbnail", "").replace("http://", "https://"), "description": vol.get("description", "Keine Beschreibung verfügbar."), "isbn": isbn, "shops": { "thalia": f"https://www.thalia.de/suche?sq={title}&partnerId={AFFILIATE_PARTNER_ID}" if not isbn else f"https://www.thalia.de/shop/{isbn}?partnerId={AFFILIATE_PARTNER_ID}", "buecher": f"https://www.buecher.de/suche/{title}", "osiander": f"https://www.osiander.de/suche/{title}" } }) return books # --------------------------- # API-Funktionen # --------------------------- def search_books(query): """Suche nach Büchern mit Google Books API.""" url = "https://www.googleapis.com/books/v1/volumes" params = { "q": query, "langRestrict": "de", "maxResults": 20, "printType": "books" } try: response = requests.get(url, params=params, timeout=10) data = response.json() except Exception as e: print("Fehler bei API-Aufruf:", e) return [] return parse_books(data) def get_bestsellers(): """Aktuelle Bestseller (deutsch).""" url = "https://www.googleapis.com/books/v1/volumes" params = { "q": "bestseller", "langRestrict": "de", "maxResults": 20, "printType": "books" } r = requests.get(url, params=params) return parse_books(r.json()) def get_trending(): """Trending (neueste Bücher).""" url = "https://www.googleapis.com/books/v1/volumes" params = { "q": "2025", # aktuelles Jahr "orderBy": "newest", "langRestrict": "de", "maxResults": 20, "printType": "books" } r = requests.get(url, params=params) return parse_books(r.json()) def get_category(cat): """Bücher einer bestimmten Kategorie.""" url = "https://www.googleapis.com/books/v1/volumes" params = { "q": f"subject:{cat}", "langRestrict": "de", "maxResults": 20, "printType": "books" } r = requests.get(url, params=params) return parse_books(r.json()) # Abgerundete weiße Karte with self.canvas.before: Color(1, 1, 1, 1) self.rect = RoundedRectangle(size=self.size, pos=self.pos, radius=[dp(12)]) self.bind(pos=self._update_rect, size=self._update_rect) # Cover (oder Platzhalter) cover_source = book_data.get("cover", "") if cover_source: self.image = AsyncImage( source=cover_source, size_hint=(None, 1), width=dp(100), allow_stretch=True, keep_ratio=True ) else: # Platzhalter self.image = BoxLayout(size_hint=(None, 1), width=dp(100)) with self.image.canvas: Color(0.9, 0.9, 0.9, 1) Rectangle(size=self.image.size, pos=self.image.pos) self.image.add_widget(Label(text="📚", font_size="40sp")) self.add_widget(self.image) # Textbereich text_box = BoxLayout(orientation="vertical", spacing=dp(4)) title = book_data.get("title", "Kein Titel") if len(title) > 45: title = title[:42] + "..." self.title_label = Label( text=title, font_size="16sp", bold=True, color=get_color_from_hex("#2c3e50"), size_hint_y=None, height=dp(35), halign="left", valign="middle", text_size=(self.width - dp(130), None) # später aktualisiert ) self.title_label.bind(size=self._update_text_size) author = book_data.get("author", "Unbekannt") self.author_label = Label( text=f"von {author}", font_size="14sp", color=get_color_from_hex("#7f8c8d"), size_hint_y=None, height=dp(25), halign="left" ) price = book_data.get("price", "0,00 €") self.price_label = Label( text=price, font_size="14sp", color=get_color_from_hex("#27ae60"), size_hint_y=None, height=dp(25), halign="left" ) text_box.add_widget(self.title_label) text_box.add_widget(self.author_label) text_box.add_widget(self.price_label) self.add_widget(text_box) def _update_rect(self, instance, value): self.rect.pos = instance.pos self.rect.size = instance.size def _update_text_size(self, instance, value): instance.text_size = (self.width - dp(130), None) def on_release(self): if self.callback: self.callback(self.book_data) # Kategorie-Menü (Bestseller, Trending, Fantasy, Thriller) menu = GridLayout(cols=2, size_hint_y=None, height=dp(100), spacing=dp(5), padding=[dp(10), 0]) btn_best = Button(text="⭐ Bestseller", background_color=get_color_from_hex("#e67e22"), color=(1,1,1,1)) btn_best.bind(on_release=lambda x: self.display_books(get_bestsellers())) btn_trend = Button(text="🔥 Trending", background_color=get_color_from_hex("#e67e22"), color=(1,1,1,1)) btn_trend.bind(on_release=lambda x: self.display_books(get_trending())) btn_fantasy = Button(text="🧙 Fantasy", background_color=get_color_from_hex("#e67e22"), color=(1,1,1,1)) btn_fantasy.bind(on_release=lambda x: self.display_books(get_category("fantasy"))) btn_thriller = Button(text="🔪 Thriller", background_color=get_color_from_hex("#e67e22"), color=(1,1,1,1)) btn_thriller.bind(on_release=lambda x: self.display_books(get_category("thriller"))) menu.add_widget(btn_best) menu.add_widget(btn_trend) menu.add_widget(btn_fantasy) menu.add_widget(btn_thriller) main_layout.add_widget(menu) # Untere Navigation nav = BoxLayout(size_hint_y=0.08, spacing=dp(2)) with nav.canvas.before: Color(1, 1, 1, 1) self.nav_rect = Rectangle(size=nav.size, pos=nav.pos) nav.bind(size=self._update_nav, pos=self._update_nav) btn_home = Button(text="🏠\nHome", font_size="12sp", background_normal='', background_color=(0,0,0,0), color=get_color_from_hex("#34495e")) btn_home.bind(on_release=lambda x: self.go_home()) btn_fav = Button(text="❤️\nFavoriten", font_size="12sp", background_normal='', background_color=(0,0,0,0), color=get_color_from_hex("#34495e")) btn_fav.bind(on_release=lambda x: self.show_favorites()) nav.add_widget(btn_home) nav.add_widget(btn_fav) main_layout.add_widget(nav) self.add_widget(main_layout) # Favoriten aus Speicher laden self.load_favorites() def _update_header(self, instance, value): self.header_rect.pos = instance.pos self.header_rect.size = instance.size def _update_nav(self, instance, value): self.nav_rect.pos = instance.pos self.nav_rect.size = instance.size def load_favorites(self): try: if store.exists("favorites"): self.favorites = store.get("favorites").get("books", []) else: self.favorites = [] except: self.favorites = [] def save_favorites(self): try: store.put("favorites", books=self.favorites) except: pass def start_search(self, instance): query = self.search_input.text.strip() if not query: return self.book_layout.clear_widgets() self.loading.opacity = 1 self.loading.value = 50 # Async-Aufruf (in echtem Projekt würde man UrlRequest verwenden; hier einfacher Thread) from kivy.clock import Clock Clock.schedule_once(lambda dt: self.do_search(query), 0.1) def do_search(self, query): books = search_books(query) self.loading.opacity = 0 self.display_books(books) def display_books(self, books): self.book_layout.clear_widgets() if not books: self.book_layout.add_widget(Label( text="Keine Bücher gefunden.\nVersuche einen anderen Suchbegriff.", size_hint_y=None, height=dp(60), color=get_color_from_hex("#7f8c8d") )) return for book in books: card = BookCard(book, callback=self.open_detail) self.book_layout.add_widget(card) def open_detail(self, book): self.manager.get_screen("detail").load_book(book) self.manager.current = "detail" def go_home(self): self.manager.current = "main" def show_favorites(self): fav_screen = self.manager.get_screen("favorites") fav_screen.load_favorites(self.favorites) self.manager.current = "favorites" #
